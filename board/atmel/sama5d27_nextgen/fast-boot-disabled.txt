# NextGen fast-U-Boot exclusions
#
# Purpose
# -------
# The chatgpt/fast-boot branch is deliberately an MMC-development bootloader:
# AT91Bootstrap -> U-Boot -> FAT on MMC -> nextgen.dtb + zImage.
#
# Keep this list when stripping features so the later NOR/NAND/MTD production
# boot path can re-enable what it actually needs instead of relying on memory.
#
# This is a tracking list, not a Kconfig fragment.

# ---------------------------------------------------------------------------
# Deferred until the NOR/NAND/MTD production boot stage
# ---------------------------------------------------------------------------

CONFIG_SPI_BOOT
# SPI boot-source support is not used by the present SD/MMC development path.

CONFIG_MTD
CONFIG_DM_MTD
CONFIG_MTD_SPI_NAND
# Generic MTD and SPI-NAND support. Revisit when kernel/DTB/rootfs move to
# production flash.

CONFIG_ATMEL_QSPI
# QSPI controller is unused by current U-Boot boot path. The U-Boot DT keeps
# the QSPI timings/pin configuration available for the later flash stage.

CONFIG_DM_SPI_FLASH
CONFIG_SPI_FLASH_SFDP_SUPPORT
CONFIG_SPI_FLASH_ATMEL
CONFIG_SPI_FLASH_MACRONIX
CONFIG_SPI_FLASH_SPANSION
CONFIG_SPI_FLASH_STMICRO
CONFIG_SPI_FLASH_SST
CONFIG_SPI_FLASH_WINBOND
CONFIG_SPI_FLASH_MTD
# SPI-NOR stack/vendor support. Current U-Boot environment is stored on the
# MMC FAT partition, so none of this is required for the fast MMC path.

CONFIG_CMD_MTD
CONFIG_CMD_MTD_NAND_READ_TEST
CONFIG_CMD_MTDPARTS
CONFIG_CMD_UBI
# Flash/MTD shell commands; re-enable only with the production flash design.

# ---------------------------------------------------------------------------
# Not used by NextGen U-Boot
# ---------------------------------------------------------------------------

CONFIG_MACB
CONFIG_NET
CONFIG_CMD_NET
# No wired Ethernet or network boot path.

CONFIG_USB
CONFIG_USB_HOST
CONFIG_DM_USB
CONFIG_USB_STORAGE
CONFIG_CMD_USB
CONFIG_USB_GADGET
CONFIG_AT91_UTMI
# No U-Boot USB host/gadget/recovery path in the fast bootloader. Linux owns
# the product USB functions later in boot.

CONFIG_CMD_SPI
# Interactive SPI command only. Keep CONFIG_SPI/CONFIG_DM_SPI and the Atmel
# SPI driver because U-Boot still initializes the ST7789 panel over SPI1.

# ---------------------------------------------------------------------------
# Command/filesystem/image helpers removed from the minimal boot surface
# ---------------------------------------------------------------------------

CONFIG_FIT
# Current image is a raw zImage plus DTB, not FIT.

CONFIG_CMD_BOOTM
# bootz is retained for the ARM Linux zImage path.

CONFIG_CMD_EXT4
# U-Boot reads only the FAT boot partition; Linux mounts the ext4 rootfs.

CONFIG_CMD_HASH
CONFIG_HASH_VERIFY
# No image hashing command/verification in the present development boot path.

CONFIG_PARTITION_UUIDS
# Boot uses fixed mmc 0:1 and /dev/mmcblk0p2, not partition UUID lookup.

CONFIG_OF_LIBFDT_OVERLAY
# U-Boot does not apply DT overlays.

CONFIG_FAT_RENAME
# FAT write support is still required by ENV_IS_IN_FAT/saveenv; rename support
# is not used by env/fat.c.

CONFIG_CMD_NVEDIT_INFO
CONFIG_CMD_BIND
CONFIG_CMD_BDI
CONFIG_CMD_MEMORY
CONFIG_CMD_LOADB
CONFIG_CMD_IMI
CONFIG_CMD_XIMG
CONFIG_CMD_EDITENV
CONFIG_CMD_EXPORTENV
CONFIG_CMD_IMPORTENV
CONFIG_CMD_ENV_EXISTS
CONFIG_CMD_CRC32
CONFIG_CMD_ECHO
CONFIG_CMD_ITEST
CONFIG_CMD_SOURCE
CONFIG_CMD_SETEXPR
CONFIG_GETOPT
# Interactive/debug helpers not needed by the automatic boot path.

CONFIG_MISC_INIT_R
# NextGen misc_init_r() was only carrying the disabled I2C-EEPROM MAC path.

# ---------------------------------------------------------------------------
# Intentionally retained on the MMC fast path
# ---------------------------------------------------------------------------

# CONFIG_MMC / CONFIG_MMC_SDHCI / CONFIG_MMC_SDHCI_ATMEL
#   Block device containing boot FAT and Linux rootfs.
# CONFIG_CMD_MMC
#   Retained for practical MMC diagnosis while this path is being developed.
# CONFIG_CMD_FAT / CONFIG_FS_FAT / CONFIG_FAT_WRITE
#   fatload for DTB/kernel and FAT-backed U-Boot environment.
# CONFIG_ENV_IS_IN_FAT / CONFIG_CMD_SAVEENV
#   Product/manufacturer environment remains persistent.
# CONFIG_HUSH_PARSER / CONFIG_CMD_RUN
#   Required by the current bootcmd first-boot env-save logic.
# CONFIG_CMD_BOOTZ
#   Boots the raw ARM zImage.
# CONFIG_SPI / CONFIG_DM_SPI / CONFIG_ATMEL_SPI
#   Required for ST7789 panel initialization on SPI1.
# CONFIG_VIDEO / CONFIG_VIDEO_ST7789_SPI / CONFIG_ATMEL_HLCD
#   Retained so panel/HLCDC are initialized before Linux/application takeover.
# CONFIG_DM_SERIAL / CONFIG_ATMEL_USART
#   Serial recovery/debug console retained during development.
